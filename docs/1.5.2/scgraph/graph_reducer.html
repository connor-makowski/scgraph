 47        """
 48        Function:
 49
 50        - Reduce the graph by bypassing pass-through nodes and summing intermediate weights.
 51        - Supports iterative reduction passes.
 52        - Sets self.reduced_graph, self.reduced_graph_connections, self.reduced_inverse_graph,
 53          self.reduced_inverse_graph_connections, self.reduced_node_chain_ids, and self.is_reduced.
 54
 55        Optional Arguments:
 56
 57        - `iterations`:
 58            - Type: int
 59            - What: Number of reduction iterations to perform.
 60              - `0`: No reduction is done.
 61              - `-1`: Reduce iteratively until convergence (no new nodes can be reduced).
 62              - `> 0`: Reduce iteratively up to the minimum of convergence and the specified count.
 63            - Default: 1
 64
 65        Returns:
 66
 67        - None
 68        """
 69        self.reset_cache()
 70        if iterations == 0:
 71            return
 72
 73        self.__ensure_inverse_graph__()
 74
 75        # 1. Identify pass-through nodes iteratively
 76        graph = self.__graph__
 77        inverse_graph = self.__inverse_graph__
 78        n = len(graph)
 79        is_reduced = [False] * n
 80
 81        current_iter = 0
 82        while True:
 83            newly_reduced_count = 0
 84            if current_iter == 0:
 85                for u in range(n):
 86                    out_dict = graph[u]
 87                    in_dict = inverse_graph[u]
 88
 89                    out_len = len(out_dict) - (1 if u in out_dict else 0)
 90                    if out_len == 1:
 91                        in_len = len(in_dict) - (1 if u in in_dict else 0)
 92                        if in_len >= 1:
 93                            is_reduced[u] = True
 94                            newly_reduced_count += 1
 95                    elif out_len == 2:
 96                        in_len = len(in_dict) - (1 if u in in_dict else 0)
 97                        if in_len > 0:
 98                            outflows = {v for v in out_dict if v != u}
 99                            inflows = {v for v in in_dict if v != u}
100                            if inflows.issubset(outflows):
101                                is_reduced[u] = True
102                                newly_reduced_count += 1
103            else:
104                # Subsequent iterations: find pass-through nodes on current reduced topology
105                for u in range(n):
106                    if is_reduced[u]:
107                        continue
108
109                    # Outflows from u through is_reduced nodes
110                    outflows = set()
111                    queue = [u]
112                    visited = {u}
113                    while queue:
114                        curr = queue.pop()
115                        if curr != u and not is_reduced[curr]:
116                            outflows.add(curr)
117                            continue
118                        for v in graph[curr]:
119                            if v not in visited:
120                                visited.add(v)
121                                queue.append(v)
122
123                    # Inflows to u through is_reduced nodes
124                    inflows = set()
125                    queue = [u]
126                    visited = {u}
127                    while queue:
128                        curr = queue.pop()
129                        if curr != u and not is_reduced[curr]:
130                            inflows.add(curr)
131                            continue
132                        for v in inverse_graph[curr]:
133                            if v not in visited:
134                                visited.add(v)
135                                queue.append(v)
136
137                    out_len = len(outflows)
138                    in_len = len(inflows)
139                    if out_len == 1:
140                        if in_len >= 1:
141                            is_reduced[u] = True
142                            newly_reduced_count += 1
143                    elif out_len == 2:
144                        if in_len > 0:
145                            if inflows.issubset(outflows):
146                                is_reduced[u] = True
147                                newly_reduced_count += 1
148
149            current_iter += 1
150
151            if newly_reduced_count == 0:
152                break
153            if iterations != -1 and current_iter >= iterations:
154                break
155
156        self.is_reduced = is_reduced
157
158        # 2. Assign chain IDs to connected components of reduced nodes
159        self.reduced_node_chain_ids = [None] * len(graph)
160        reduced_node_chain_ids = self.reduced_node_chain_ids
161        current_chain_id = 0
162        for u in range(len(graph)):
163            if is_reduced[u] and reduced_node_chain_ids[u] is None:
164                queue = [u]
165                reduced_node_chain_ids[u] = current_chain_id
166                while queue:
167                    curr = queue.pop()
168                    for v in graph[curr]:
169                        if (
170                            v != curr
171                            and is_reduced[v]
172                            and reduced_node_chain_ids[v] is None
173                        ):
174                            reduced_node_chain_ids[v] = current_chain_id
175                            queue.append(v)
176                    for v in inverse_graph[curr]:
177                        if (
178                            v != curr
179                            and is_reduced[v]
180                            and reduced_node_chain_ids[v] is None
181                        ):
182                            reduced_node_chain_ids[v] = current_chain_id
183                            queue.append(v)
184                current_chain_id += 1
185
186        # 3. Build reduced graph and connections (outbound from all nodes)
187        self.reduced_graph = [{} for _ in range(len(graph))]
188        self.reduced_graph_connections = [None] * len(graph)
189        reduced_graph = self.reduced_graph
190        reduced_graph_connections = self.reduced_graph_connections
191
192        for A in range(len(graph)):
193            best_dist = {}
194            open_leaves = [(0, A, [])]
195
196            while open_leaves:
197                dist, u, path = heappop(open_leaves)
198                if u in best_dist and best_dist[u] <= dist:
199                    continue
200                best_dist[u] = dist
201
202                if u != A and not is_reduced[u]:
203                    # Boundary non-reduced node. Record connection.
204                    reduced_graph[A][u] = dist
205                    if path:
206                        if reduced_graph_connections[A] is None:
207                            reduced_graph_connections[A] = {}
208                        reduced_graph_connections[A][u] = path
209                    continue
210
211                for v, w in graph[u].items():
212                    new_dist = dist + w
213                    if v not in best_dist or new_dist < best_dist[v]:
214                        new_path = path if u == A else path + [u]
215                        heappush(open_leaves, (new_dist, v, new_path))
216
217        # 4. Build reduced inverse graph and connections (inbound into all nodes)
218        self.reduced_inverse_graph = [{} for _ in range(len(graph))]
219        self.reduced_inverse_graph_connections = [None] * len(graph)
220        reduced_inverse_graph = self.reduced_inverse_graph
221        reduced_inverse_graph_connections = (
222            self.reduced_inverse_graph_connections
223        )
224
225        for B in range(len(graph)):
226            best_dist = {}
227            open_leaves = [(0, B, [])]
228
229            while open_leaves:
230                dist, u, path = heappop(open_leaves)
231                if u in best_dist and best_dist[u] <= dist:
232                    continue
233                best_dist[u] = dist
234
235                if u != B and not is_reduced[u]:
236                    # Boundary non-reduced node reaching B
237                    reduced_inverse_graph[B][u] = dist
238                    fwd_path = list(reversed(path))
239                    if fwd_path:
240                        if reduced_inverse_graph_connections[B] is None:
241                            reduced_inverse_graph_connections[B] = {}
242                        reduced_inverse_graph_connections[B][u] = fwd_path
243                    continue
244
245                for v, w in inverse_graph[u].items():
246                    new_dist = dist + w
247                    if v not in best_dist or new_dist < best_dist[v]:
248                        new_path = path if u == B else path + [u]
249                        heappush(open_leaves, (new_dist, v, new_path))

255        - Wrap a CH/TNR heuristic function to contract reduced nodes before non-reduced nodes.

269    def run_with_reduced(
270        self, func, *args, is_bidirectional: bool = False, **kwargs
271    ):
272        """
273        Function:
274
275        - Run a graph routing function on the reduced graph.
276        - Same-chain queries route directly on the unreduced graph.
277        - Bidirectional algorithms route on reduced_graph and reduced_inverse_graph.
278        - One-sided algorithms with reduced destination solve via boundary entry nodes from reduced_inverse_graph.
279
280        Required Arguments:
281
282        - `func`
283            - Type: function
284            - What: The routing algorithm function to execute
285
286        Returns:
287
288        - The return value of the wrapped routing algorithm (typically a dict with 'path' and 'length')
289        """
290        origin_id = kwargs.get("origin_id")
291        if origin_id is None and len(args) > 0:
292            origin_id = args[0]
293
294        destination_id = kwargs.get("destination_id")
295        if destination_id is None and len(args) > 1:
296            destination_id = args[1]
297
298        # 1. Same-chain check: solve on unreduced graph with dijkstra
299        if destination_id is not None and self.is_same_chain(
300            origin_id, destination_id
301        ):
302            length_only = kwargs.get("length_only", False)
303            res = GraphReducer.dijkstra_on_graph(
304                self.unreduced_graph, origin_id, destination_id
305            )
306            if length_only:
307                res.pop("path", None)
308            return res
309
310        is_reduced = self.is_reduced
311
312        # 2. Bidirectional algorithms, Contraction Hierarchies, and Transit Node Routing
313        is_bidirectional = (
314            is_bidirectional
315            or getattr(func, "__bidirectional__", False)
316            or func.__name__
317            in (
318                "bidirectional_dijkstra",
319                "contraction_hierarchy",
320                "tnr",
321            )
322        )
323        if is_bidirectional:
324            res = func(self, *args, **kwargs)
325            if isinstance(res, dict) and "path" in res:
326                res["path"] = self.expand_path(res["path"])
327            return res
328
329        # 3. One-sided algorithms with unreduced destination (or None)
330        if destination_id is None or not is_reduced[destination_id]:
331            res = func(self, *args, **kwargs)
332            if isinstance(res, dict) and "path" in res:
333                res["path"] = self.expand_path(res["path"])
334            return res
335
336        # 4. One-sided algorithm with reduced destination
337        entry_nodes = self.reduced_inverse_graph[destination_id]
338        if not entry_nodes:
339            raise Exception(
340                "Something went wrong, the origin and destination nodes are not connected."
341            )
342
343        best_length = float("inf")
344        best_res = None
345        best_entry = None
346
347        for entry_u, entry_dist in entry_nodes.items():
348            try:
349                call_kwargs = dict(kwargs)
350                call_args = list(args)
351                if "destination_id" in call_kwargs:
352                    call_kwargs["destination_id"] = entry_u
353                elif len(call_args) > 1:
354                    call_args[1] = entry_u
355
356                res_u = func(self, *call_args, **call_kwargs)
357                total_dist = res_u["length"] + entry_dist
358                if total_dist < best_length:
359                    best_length = total_dist
360                    best_res = res_u
361                    best_entry = entry_u
362            except Exception:
363                continue
364
365        if best_res is None or best_length == float("inf"):
366            raise Exception(
367                "Something went wrong, the origin and destination nodes are not connected."
368            )
369
370        if "path" in best_res:
371            expanded_path = self.expand_path(best_res["path"])
372            tail = []
373            if (
374                self.reduced_inverse_graph_connections is not None
375                and self.reduced_inverse_graph_connections[destination_id]
376                is not None
377            ):
378                tail = self.reduced_inverse_graph_connections[
379                    destination_id
380                ].get(best_entry, [])
381            full_path = expanded_path + tail + [destination_id]
382            return {
383                "path": full_path,
384                "length": best_length,
385            }
386        return {
387            "length": best_length,
388        }

393    def dijkstra_on_graph(
394        graph: list[dict[int, float]],
395        origin_id: int | set[int],
396        destination_id: int,
397    ) -> dict:
398        """
399        Function:
400
401        - Internal Dijkstra solver on a specified graph dictionary list (used for same-chain routing on unreduced graph).
402        """
403        origin_ids = {origin_id} if isinstance(origin_id, int) else origin_id
404        distance_matrix = [float("inf")] * len(graph)
405        predecessor = [-1] * len(graph)
406        open_leaves = []
407
408        for oid in origin_ids:
409            distance_matrix[oid] = 0
410            heappush(open_leaves, (0, oid))
411
412        while open_leaves:
413            current_distance, current_id = heappop(open_leaves)
414            if current_id == destination_id:
415                break
416            if current_distance == distance_matrix[current_id]:
417                for (
418                    connected_id,
419                    connected_distance,
420                ) in graph[current_id].items():
421                    possible_distance = current_distance + connected_distance
422                    if possible_distance < distance_matrix[connected_id]:
423                        distance_matrix[connected_id] = possible_distance
424                        predecessor[connected_id] = current_id
425                        heappush(open_leaves, (possible_distance, connected_id))
426        if current_id != destination_id:
427            raise Exception(
428                "Something went wrong, the origin and destination nodes are not connected."
429            )
430
431        path = []
432        curr = destination_id
433        while curr != -1:
434            path.append(curr)
435            curr = predecessor[curr]
436        path.reverse()
437
438        return {
439            "path": path,
440            "length": distance_matrix[destination_id],
441        }

445    def expand_path(self, path: list[int]) -> list[int]:
446        """
447        Function:
448
449        - Expand a path of reduced graph nodes to restore all intermediate pass-through nodes.
450
451        Required Arguments:
452
453        - `path`
454            - Type: list[int]
455            - What: The path in the reduced graph
456
457        Returns:
458
459        - The fully expanded path matching the original graph
460        """
461        fwd_conns = getattr(self, "reduced_graph_connections", None)
462        inv_conns = getattr(self, "reduced_inverse_graph_connections", None)
463        if fwd_conns is None and inv_conns is None:
464            return path
465        new_path = []
466        append = new_path.append
467        extend = new_path.extend
468        for i in range(len(path) - 1):
469            u = path[i]
470            v = path[i + 1]
471            append(u)
472            expanded = False
473            if fwd_conns is not None and fwd_conns[u] is not None:
474                pass_throughs = fwd_conns[u].get(v)
475                if pass_throughs:
476                    extend(pass_throughs)
477                    expanded = True
478            if (
479                not expanded
480                and inv_conns is not None
481                and inv_conns[v] is not None
482            ):
483                pass_throughs = inv_conns[v].get(u)
484                if pass_throughs:
485                    extend(pass_throughs)
486        if path:
487            append(path[-1])
488        return new_path

490    def is_same_chain(
491        self,
492        origin_id: int | set[int] | list[int],
493        destination_id: int | None,
494    ) -> bool:
495        """
496        Function:
497
498        - Check whether any origin node and the destination node belong to the same reduced chain.
499
500        Required Arguments:
501
502        - `origin_id`
503            - Type: int | set[int] | list[int]
504            - What: The id(s) of the origin node(s)
505        - `destination_id`
506            - Type: int | None
507            - What: The id of the destination node
508
509        Returns:
510
511        - True if both origin and destination are reduced nodes and share the same chain_id, False otherwise.
512        """
513        reduced_node_chain_ids = getattr(self, "reduced_node_chain_ids", None)
514        if reduced_node_chain_ids is None or destination_id is None:
515            return False
516        if destination_id < 0 or destination_id >= len(reduced_node_chain_ids):
517            return False
518        dest_chain = reduced_node_chain_ids[destination_id]
519        if dest_chain is None:
520            return False
521        if isinstance(origin_id, int):
522            if 0 <= origin_id < len(reduced_node_chain_ids):
523                return reduced_node_chain_ids[origin_id] == dest_chain
524            return False
525        for oid in origin_id:
526            if (
527                0 <= oid < len(reduced_node_chain_ids)
528                and reduced_node_chain_ids[oid] == dest_chain
529            ):
530                return True
531        return False

394    def dijkstra_on_graph(
395        graph: list[dict[int, float]],
396        origin_id: int | set[int],
397        destination_id: int,
398    ) -> dict:
399        """
400        Function:
401
402        - Internal Dijkstra solver on a specified graph dictionary list (used for same-chain routing on unreduced graph).
403        """
404        origin_ids = {origin_id} if isinstance(origin_id, int) else origin_id
405        distance_matrix = [float("inf")] * len(graph)
406        predecessor = [-1] * len(graph)
407        open_leaves = []
408
409        for oid in origin_ids:
410            distance_matrix[oid] = 0
411            heappush(open_leaves, (0, oid))
412
413        while open_leaves:
414            current_distance, current_id = heappop(open_leaves)
415            if current_id == destination_id:
416                break
417            if current_distance == distance_matrix[current_id]:
418                for (
419                    connected_id,
420                    connected_distance,
421                ) in graph[current_id].items():
422                    possible_distance = current_distance + connected_distance
423                    if possible_distance < distance_matrix[connected_id]:
424                        distance_matrix[connected_id] = possible_distance
425                        predecessor[connected_id] = current_id
426                        heappush(open_leaves, (possible_distance, connected_id))
427        if current_id != destination_id:
428            raise Exception(
429                "Something went wrong, the origin and destination nodes are not connected."
430            )
431
432        path = []
433        curr = destination_id
434        while curr != -1:
435            path.append(curr)
436            curr = predecessor[curr]
437        path.reverse()
438
439        return {
440            "path": path,
441            "length": distance_matrix[destination_id],
442        }

447        """
448        Function:

461        """
462        fwd_conns = getattr(self, "reduced_graph_connections", None)
463        inv_conns = getattr(self, "reduced_inverse_graph_connections", None)
464        if fwd_conns is None and inv_conns is None:

467        append = new_path.append
468        extend = new_path.extend

486                    extend(pass_throughs)
